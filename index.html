Run this standalone Node.js Express server to shield your Apify key and add in-memory caching to save budget.

/**
 * SECURE APIFY WEATHER PROXY SERVER (Node.js + Express)
 * ----------------------------------------------------
 * Dependencies:
 *   npm install express cors dotenv node-cache express-rate-limit node-fetch
 */

const express = require('express');
const cors = require('cors');
const NodeCache = require('node-cache');
const rateLimit = require('express-rate-limit');
require('dotenv').config();

const app = express();
const PORT = process.env.PORT || 4000;

// 1. In-memory cache: Stores weather data for 15 minutes (900 seconds)
// Saves Apify compute units by preventing duplicate actor runs for the same city
const weatherCache = new NodeCache({ stdTTL: 900 });

// 2. Strict CORS & Rate Limiting
app.use(cors({ origin: '*' })); // Configure your frontend domain here in prod
app.use(express.json());

const limiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 60, // Limit each IP to 60 requests per window
  message: { error: 'Too many requests. Please wait a moment.' }
});
app.use('/api/', limiter);

// 3. Environment Variables (Secret Token NEVER sent to browser)
const APIFY_TOKEN = process.env.APIFY_TOKEN;
const APIFY_ACTOR_ID = process.env.APIFY_ACTOR_ID || 'your-actor-id';

app.get('/api/weather', async (req, res) => {
  try {
    const city = req.query.city?.trim();
    if (!city) {
      return res.status(400).json({ error: 'City query parameter is required.' });
    }

    const cacheKey = city.toLowerCase();
    const cachedData = weatherCache.get(cacheKey);
    if (cachedData) {
      return res.json({ source: 'proxy-cache', data: cachedData });
    }

    // Call Apify Actor synchronously via official run-sync endpoint
    // https://api.apify.com/v2/acts/{ACTOR_ID}/run-sync-get-dataset-items?token={TOKEN}
    const apifyUrl = `https://api.apify.com/v2/acts/${APIFY_ACTOR_ID}/run-sync-get-dataset-items?token=${APIFY_TOKEN}`;
    
    const response = await fetch(apifyUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ city: city, units: req.query.units || 'metric' })
    });

    if (!response.ok) {
      throw new Error(`Apify returned status ${response.status}`);
    }

    const result = await response.json();
    // Normalize response dataset item (handles both array and single object)
    const weatherData = Array.isArray(result) ? result[0] : result;

    // Cache the response
    weatherCache.set(cacheKey, weatherData);

    return res.json({ source: 'apify-fresh', data: weatherData });
  } catch (error) {
    console.error('Weather Proxy Error:', error.message);
    return res.status(500).json({ error: 'Failed to retrieve weather data from Apify.' });
  }
});

app.listen(PORT, () => {
  console.log(`🌤️ Weather Proxy listening safely on port ${PORT}`);
});

Why do you need a proxy for Apify?

Apify actors consume credits every time they run. If you embed your secret APIFY_TOKEN directly into your web dashboard's JavaScript, any user can inspect network traffic, extract your token, and deplete your Apify billing limits.

1. Hide Secret Keys Your token resides solely inside .env on the proxy server. The client never sees it.
2. Cost Caching Multiple users checking "Kyoto" in 15 minutes receive the cached copy instead of running multiple Apify runs.
3. Rate Limiting Built-in IP throttling prevents bots and scrapers from exhausting your quota.
